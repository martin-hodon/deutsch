# HOW POLISH SOUNDS (engine/speak.js loadProsody): the format is de/rules/speech.txt's.
rhythm = syllable      -- Polish keeps its vowels full and its syllables about even; the stress is on the second-last syllable of every word
even = 0.5
long = 1               -- no vowel length (a ː is not written in Polish IPA)
stress = 1.2           -- the stressed vowel a little longer
unstressed = 1         -- unstressed vowels keep their quality (no reduction)
final = 1.2            -- the last vowel of what is said drawn out
pause = 35             -- ms between words
slow = 1.15            -- a little slower than conversation (the listener's own tempo is set in the trainer)
accent = 1.08          -- a modest pitch rise on the stressed syllable
range = 0.15
aspiration = 0.15      -- p t k are not aspirated
