# HOW GERMAN SOUNDS (engine/speak.js loadProsody): the rhythm and timing of the voice -- what differs between the
# languages; the sounds themselves are shared. `key = value -- note`; `say X = Y`: the symbol X of this language's IPA
# is the sound Y.
rhythm = stress        -- stress-timed: the stressed syllables come at even intervals, the unstressed ones between them are squeezed
even = 0.35            -- how far the timing is evened out (0 none, 1 feet of equal length) -- part of the way, as speakers do
long = 1.7             -- a long vowel (ː: Staat) against a short one (Stadt) under stress -- German's length is a phoneme, kept wide; unstressed much less
stress = 1.2           -- a stressed vowel's length
unstressed = 0.9       -- an unstressed vowel's (the schwa has its own, shorter)
final = 1.15           -- the last vowel of what is said, lengthened
pause = 45             -- ms between words: a short gap only (longer ones sound like a list)
slow = 1.15             -- everything this much slower than conversation, for a learner
accent = 1.08          -- the pitch on a stressed vowel, times the baseline -- a little: German speech is fairly level
range = 0.15           -- the pitch's fall over a statement, as a share of the voice's pitch
aspiration = 1         -- p t k before a vowel are aspirated in full
