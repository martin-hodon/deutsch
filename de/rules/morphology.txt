# ---- LETTERS: named groups for the rules -- {name} in a pattern is any letter of the group (as [abc] is any of a, b,
# c); a group may be made of others: {a} - {b} - [cd] (the letters of a, not of b, not c or d)

group.consonant = b c d f g h j k l m n p q r s t v w x z ß
group.liquid = l r
group.nasal = m n
group.s-sound = s ß x z
# the consonants that want an e between them and m/n before -st, -t (rechnest, atmet, öffnet) -- not h, silent in
# wohnen; nor a liquid or a nasal (filmt, lernt, kämmst); ch is written as it is
group.obstruent = {consonant} - {liquid} - {nasal} - [hj]
# a named PATTERN, in a rule as {name} too: a stem that takes an e before -st, -t, -te (arbeitest, atmet, rechnete)

# ---- RULES: a form made by replacing a word's end -- `rule.KIND pattern /sound/ = -cut > ending | ending ...`
# `rule.KIND` with no pattern is the STANDARD; the others are its EXCEPTIONS, tried in their order before it: the first
# whose pattern ends the word (and whose /sound/ ends its IPA, when given; and whose cut ends it) cuts `cut` and puts
# the ending (`umlaut` before the cut: the stem's last vowel that can takes its umlaut first) -- one per person (1sg
# 2sg 3sg 1pl 2pl 3pl) or one for every form; `-` nothing; `.` as the next rule that
# fits the word says it (an exception writes only where it differs). A pattern is read as a regular expression: [abc]
# one of these letters, {group} or {pattern}, (a|b) either. (Every verb ends in -en, -ln or -rn but tun -- and sein,
# all its own paradigms: [lru]n.) The sounds: a cut's or an ending's own
# [ipa] if written, else its ipa.* above, else its letters (e as ə, r as ʁ). What a cut and its ending share stays
# (-eln > elst is -n > st).

# the PRESENT, from the infinitive
rule.present = -en > e | st | t | en | t | en
rule.present {consonant}ien = -n [ən] > - | st | t | n [ən] | t | n [ən]
rule.present {e-insert}en = -en > . | est | et | . | et | .
rule.present {s-sound}en = -en > . | t | . | . | . | .
rule.present eln = -eln > le | . | . | . | . | .
rule.present [lru]n = -n > e | st | t | n | t | n

# verb's (brachte, dachte) from its own past
rule.past = -en > te | test | te | ten | tet | ten
rule.past {consonant}ien = -n [ən] > te | test | te | ten | tet | ten
rule.past {e-insert}en = -en > ete | etest | ete | eten | etet | eten
rule.past [lru]n = -n > te | test | te | ten | tet | ten
rule.past-strong = > - | st | - | en | t | en
rule.past-strong {e-insert} = > . | est | . | . | et | .
rule.past-strong {s-sound} = > . | est | . | . | . | .
rule.past-strong e = > . | . | . | n [ən] | . | n [ən]
rule.past-mixed = -te > te | test | te | ten | tet | ten
#   (du fandest, ihr fandet; du lasest; wir schrien; du brachtest)

# the PARTICIPLE's ending (its ge- by the stress: the code), from the infinitive
rule.participle = -en > t
rule.participle {consonant}ien = -n [ən] > t
rule.participle {e-insert}en = -en > et
rule.participle [lru]n = -n > t

# KONJUNKTIV I, from the infinitive (er sei: paradigm.sein.subj1)
rule.subj1 = -en > e | est | e | en | et | en
rule.subj1 {consonant}ien = -n [ən] > - | st | - | n [ən] | t | n [ən]
rule.subj1 eln = -eln > le | . | le | . | . | .
rule.subj1 [lr]n = -n > e | st | e | n | t | n
rule.subj1 un = -n > e | est | e | en | et | en
#   (er komme; er knie; er sammle, du sammelst; er wandere, ihr wandert; er tue)

# KONJUNKTIV II, from the dictionary's past with the umlaut (kam -> käme, fand -> fände, ging -> ginge, sah -> sähe; a
# mixed verb's: brachte -> brächte) -- for the verbs that keep their own in everyday German; the rest: würde + infinitive
verb.subj2-own = kommen | gehen | geben | bleiben | lassen | finden | liegen | sitzen | stehen | tun | halten | heißen | sehen | ziehen | bringen | denken
rule.subj2 = umlaut > e | est | e | en | et | en
rule.subj2-mixed = umlaut -te > te | test | te | ten | tet | ten

# the IMPERATIVE for du, from the infinitive (a changed stem -- sprich! -- from the dictionary: the code)
rule.imperative = -en > -
rule.imperative {consonant}ien = -n [ən] > -
rule.imperative {e-insert}en = -en > e
rule.imperative eln = -eln > le
rule.imperative ern = -n > e
rule.imperative un = -n > -
#   (komm! knie! arbeite! atme! sammle! wandere! tu!)

# an ADJECTIVE's stem before an ending (no standard: the adjective as it is)
rule.adj-stem e /ə/ = -e > -
rule.adj-stem el /əl/ = -el > l
rule.adj-stem uer /ɐ/ = -er > r
#   (leise: leise; dunkel: dunkle -- not parallel [eːl], viel; teuer: teure, sauer: saure -- -uer is au/eu + er, a
