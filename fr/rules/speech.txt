# HOW FRENCH SOUNDS (engine/speak.js loadProsody): France French -- the format is de/rules/speech.txt's.
rhythm = syllable      -- syllable-timed: even syllables, the last one of a phrase lengthened (French stress is final)
even = 0.6
long = 1.2             -- a ː is rare in French IPA (it marks a vowel before a final r or z: rose); a little longer only
stress = 1.15          -- the stress mark falls on a word's last syllable: lengthened, not louder
unstressed = 1         -- every vowel keeps its full quality
final = 1.4            -- the last vowel of what is said is drawn out: the French phrase ends long
pause = 15             -- ms between words: French links everything (liaison, enchaînement) -- hardly any pause
slow = 1.15            -- a little slower than conversation (the listener's own tempo is set in the trainer)
accent = 1.04          -- the pitch moves little within a word; a phrase rises at its end and falls at the sentence's
range = 0.12           -- a flat declination
aspiration = 0.15      -- p t k are not aspirated
