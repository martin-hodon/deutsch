# THE CATEGORIES (sentences/README.md): what a word's meaning IS, as a graph. A line is
#     name : parent, parent -- what it is (examples)
# a category may have several parents (a meal is food and an event); every category leads up to `entity`. A word belongs
# to categories (sentences/meaning/), and a verb or an adjective asks for categories (eat: an `animal` eats `food`) --
# a word fits when one of its categories is, or lies under, the one asked for.

# A category may be NAMED by a concept: `mammal : beast = cicavec` (`= POS id`, N when no POS; an affordance by its verb,
# `openable : concrete = V otvoriť`) -- the dictionary shows the category by that concept's word in the language, and
# that word's entry lists the category's direct members.
#
# DIRECTIVES (tools/meaning.py and trainer/meaning.js read them):
#   BUCKET: categories too mixed to be a verb's subject or slot alone -- such a frame names a subcategory, adds an
#           affordance (+x) or lists its nouns ({Geburtstag|Hochzeit}); the checker says where
#   CLOSED: categories an adjective reaches only by naming them (or one under them): "alt : person, object" is said of
#           people, "schwer : object, animal" is not ("schwerer Nachbar")
#   AFFORDANCE: the categories that say what one can do with a thing (below), not what it is
BUCKET: event activity state idea problem information phenomenon moment writing creative-work object
CLOSED: person professional role text document money illness feeling
AFFORDANCE: openable lockable breakable repairable washable cleanable sittable drivable rideable playable cookable hangable plantable burnable foldable cuttable switchable enterable climbable fillable attendable

entity -- anything at all
concrete : entity -- what is there physically, can be seen or touched
abstract : entity -- what is not physical: times, events, states, ideas, qualities
# (inanimate: what a colour, a size, a texture is said of -- not a person: no "dunkelgrüner Angehöriger")
inanimate : concrete -- what is physical and not alive: things, stuff, places, the sky, the weather

# ---- the living
living : concrete -- what lives and grows
agent : entity -- what can act on purpose: people, groups of them, institutions
animal : living -- any animal (people are animals too: a slot for `animal` takes people)
person : animal, agent = človek -- a human being (Mann, Frau, Freund)
beast : animal = zviera -- an animal that is no person (Tier, Hund, Vogel, Fisch): `animal` includes people, `beast` does not
child : person = dieťa -- a young person (Kind, Baby, Junge)
infant : child = bábätko -- a baby (Baby, Säugling, Kleinkind): as a subject only of what grammar.txt infant.verbs lists
relative : person = príbuzný -- a family member (Mutter, Bruder, Onkel)
professional : person -- a person by their work or role (Arzt, Lehrer, Polizist, König)
role : person -- a person by what they do or where they are at a time (Kunde, Besucher, Nutzer, Teilnehmer, Fahrer, Gast): no looks, no fame of their own
national : person -- a person by nationality or origin (Engländer, Kroatin, Europäer, Berliner)
group : agent, concrete = skupina -- several people as one (Familie, Team, Volk, Publikum, Armee)
institution : agent, abstract = inštitúcia -- an organisation that acts (Firma, Regierung, Universität, Partei, Bank as a firm)
mammal : beast = cicavec -- (Hund, Pferd, Kuh, Maus)
bird : beast = vták -- (Vogel, Huhn, Adler)
fish-animal : beast = ryba -- a fish alive in the water (Fisch, Hai)
insect : beast = hmyz -- also spiders, worms (Biene, Fliege, Spinne)
reptile : beast -- also amphibians (Schlange, Frosch)
pet : beast = domáce zviera -- kept in a home (Hund, Katze)
livestock : beast = hospodárske zviera -- kept on a farm (Kuh, Schwein, Huhn)
plant : living = rastlina -- (Pflanze, Gras)
fungus : living = huba -- (Pilz, Schimmel)
being : agent, concrete = bytosť -- a supernatural being (Gott, Engel, Geist, Teufel, Drache)
tree : plant = strom -- (Baum, Eiche)
flower : plant = kvet -- (Blume, Rose)
crop : plant -- grown in a field for food or use (Weizen, Kartoffel as a plant)
plant-part : concrete -- (Blatt, Wurzel, Ast)
body-part : concrete = časť tela -- a part of a body (Kopf, Hand, Herz, Haar)

# ---- food and drink
consumable : inanimate -- what is eaten or drunk
food : consumable = jedlo -- what is eaten
drink : consumable, liquid = nápoj -- what is drunk (Wasser, Bier, Saft, Milch)
fruit : food = ovocie -- (Apfel, Banane)
vegetable : food = zelenina -- (Karotte, Tomate)
meat : food = mäso -- also fish as food (Fleisch, Wurst, Schinken)
baked-goods : food = pečivo -- (Brot, Kuchen, Brötchen)
sweets : food = sladkosť -- (Schokolade, Bonbon)
dairy : food = mliečny výrobok -- (Käse, Butter, Joghurt)
grain : food -- (Reis, Mehl, Nudeln)
seasoning : food = korenie -- (Salz, Pfeffer, Zucker)
dish : food = pokrm -- something cooked, a course (Suppe, Salat, Braten)
meal : food, event -- (Frühstück, Mittagessen)
medicine : consumable = liek -- (Medikament, Tablette)

# ---- matter
substance : inanimate = látka -- stuff, not a thing (Wasser, Luft, Sand)
liquid : substance -- (Öl, Blut)
gas : substance = plyn -- (Luft, Dampf)
material : substance = materiál -- what things are made of
metal : material = kov -- (Eisen, Gold)
wood : material = drevo
stone : material = kameň -- also a stone as a thing (Stein, Felsen)
fabric : material -- (Stoff, Wolle, Leder)
paper : material = papier
fuel : substance = palivo -- (Benzin, Kohle)

# ---- things made
object : inanimate = predmet -- a physical object, of any size (Gegenstand; a car, a fridge, a door)
thing : object = vec -- an object small enough to pick up and carry (Ding, Stein, Ball, Tasse, Messer)
artifact : object = výrobok -- made by people
tool : artifact, thing = nástroj -- (Hammer, Messer, Schlüssel, Besen)
device : artifact = prístroj -- a machine or an apparatus (Maschine, Computer, Telefon, Lampe)
vehicle : artifact = vozidlo -- (Auto, Bus, Fahrrad, Schiff, Flugzeug)
road-vehicle : vehicle -- one on the road, an everyday sight (Auto, Bus, Fahrrad, Taxi, Motorrad)
furniture : artifact = nábytok -- (Tisch, Stuhl, Bett)
# (vehicles, furniture, devices, building parts: objects, not things -- a device one carries, Handy, is a `thing` by its
#  own noun line)
wearable : thing -- what is worn on the body
clothing : artifact, wearable = oblečenie -- (Hemd, Hose, Schuh, Hut)
jewelry : artifact, wearable, thing = šperk -- (Ring, Kette)
container : artifact, thing -- what holds things (Tasse, Flasche, Kiste, Tasche, Topf)
kitchenware : artifact, thing = riad -- (Teller, Löffel, Pfanne)
instrument : artifact = hudobný nástroj -- a musical instrument (Gitarre, Klavier)
weapon : artifact, thing = zbraň -- (Schwert, Gewehr)
toy : artifact, thing = hračka -- also sport equipment (Ball, Puppe)
text : artifact, thing = text -- a written thing to be read (Buch, Brief, Zeitung, Zettel; what it SAYS is information)
picture : artifact, thing = obraz -- (Bild, Foto, Karte)
money : artifact, thing = peniaze -- (Geld, Münze, Schein)
document : artifact, thing = dokument -- a paper that proves, permits or binds (Pass, Ausweis, Visum, Ticket, Gutschein, Vertrag, Formular)
building-part : artifact -- (Dach, Wand, Tür, Fenster, Treppe, Boden as a floor)

# ---- places
place : inanimate = miesto (priestor) -- where one can be or go
building : place = budova -- (Haus, Kirche, Schule as a building; not a `thing`: one does not carry or wipe it)
room : place = miestnosť -- a part of a building (Küche, Zimmer)
shop : building = obchod -- where one buys (Laden, Bäckerei, Supermarkt)
settlement : place -- (Stadt, Dorf)
country : place = krajina (štát) -- also a state, a continent (Deutschland, Europa)
region : place = región -- (Gegend, Land as countryside)
landscape : place = krajina (prírodná) -- natural ground (Wald, Wiese, Berg, Strand)
water-body : place -- (See, Fluss, Meer)
road : place = cesta (komunikácia) -- (Straße, Weg, Brücke)
celestial : inanimate = nebeské teleso -- (Sonne, Mond, Stern)
phenomenon : inanimate = jav -- what happens in nature (Regen, Wind, Feuer, Gewitter)
sound : phenomenon = zvuk -- (Geräusch, Lärm, Klang, Ton, Knall)

# ---- the abstract
time : abstract = čas (trvanie) -- (Zeit)
period : time = obdobie -- a stretch of time (Jahr, Woche, Sommer, Stunde)
moment : time = okamih -- a point in time (Augenblick, Termin)
event : abstract = udalosť -- what happens, an occasion (Treffen, Reise, Konzert)
incident : event = nehoda -- what befalls someone (Unfall, Katastrophe, Explosion, Brand)
process : abstract = proces -- a change that goes on (Entwicklung, Änderung, Wachstum, Verfall, Rückgang)
activity : event = činnosť -- what is done (Arbeit, Reise, Kauf)
game : activity = hra -- a game or a sport (Spiel, Fußball, Schach, Tennis)
crime : activity = zločin -- (Diebstahl, Mord, Betrug)
celebration : event = oslava -- (Fest, Party, Hochzeit, Geburtstag)
state : abstract = stav -- how someone or something is (Gesundheit, Frieden, Ruhe)
illness : state = choroba -- (Krankheit, Grippe, Schmerz)
feeling : state = cit -- (Freude, Angst, Liebe, Wut)
quality : abstract = vlastnosť -- a property (Größe, Farbe, Schönheit)
ability : quality = schopnosť -- what someone can do (Fähigkeit, Talent, Kraft)
quantity : abstract = množstvo -- (Menge, Zahl, Teil)
measure : abstract = jednotka -- a unit (Meter, Kilo, Liter)
currency : measure = mena -- a unit of money (Euro, Cent, Dollar): what one pays, earns, owes
information : abstract = informácia -- what is known or told (Nachricht, Wissen, Antwort)
idea : abstract = myšlienka -- what is thought (Gedanke, Plan, Idee, Meinung)
language : abstract = jazyk (reč) -- a language (Sprache, Deutsch as a language)
writing : abstract = písmo -- written signs and units of language (Wort, Satz, Schrift, Alphabet, Buchstabe, Überschrift)
software : abstract = softvér -- (Programm, App, Software, Datei, Website)
creative-work : abstract = dielo -- a work of art (Lied, Film, Gedicht, Geschichte)
rule : abstract = pravidlo -- (Gesetz, Regel)
problem : abstract = problém -- (Problem, Frage, Schwierigkeit)
relation : abstract = vzťah -- (Freundschaft, Ehe, Kontakt)
direction : abstract = smer -- (Norden, Richtung)
position : abstract = poloha -- a point or part of space as such (Stelle, Seite, Mitte, Nähe, Lage): not a place one is in or goes to

# ---- AFFORDANCES: what one can do with it -- categories a noun has besides what it IS (a door IS a building-part and
# is openable, lockable); a verb's slot asks for one with `+x` (V zumachen : person > A object +openable). They cut
# across the tree, so they sit right under `concrete` and are never what an adjective or a frame names alone.
openable : concrete = V otvoriť -- one opens and closes it (Tür, Fenster, Flasche, Kiste, Buch, Augen, Dose)
lockable : concrete = V zamknúť -- one locks it (Tür, Auto, Schrank, Fahrrad, Koffer)
breakable : concrete = V rozbiť -- it breaks, shatters (Glas, Tasse, Teller, Fenster, Spiegel, Bein, Knochen)
repairable : concrete = V opraviť -- one repairs it (Auto, Fahrrad, Uhr, Dach, Computer, Schuh, Heizung)
washable : concrete = V umyť -- one washes it (Hemd, Hände, Haare, Auto, Teller, Gemüse)
cleanable : concrete = V čistiť -- one cleans, wipes, sweeps it (Boden, Küche, Fenster, Tisch, Zimmer, Bad)
sittable : concrete = V sedieť -- one sits on it (Stuhl, Sofa, Bank, Bett, Sessel, Hocker, Boden, Rasen)
drivable : concrete = V riadiť -- one drives, steers, parks it (Auto, Bus, Lkw, Traktor, Zug, Motorrad)
rideable : concrete = V jazdiť -- one rides it (Pferd, Fahrrad, Motorrad, Esel, Kamel)
playable : concrete = V hrať -- one plays it (Gitarre, Klavier, Spiel, Fußball, Schach, Karten, Lied)
cookable : concrete = V variť -- one cooks, fries, bakes, boils it (Kartoffel, Fleisch, Nudeln, Reis, Ei, Suppe, Kuchen)
hangable : concrete = V zavesiť -- one hangs it up (Bild, Mantel, Jacke, Lampe, Vorhang, Wäsche, Plakat)
plantable : concrete = V sadiť -- one plants, waters it (Baum, Blume, Kartoffel, Samen, Rose, Strauch)
burnable : concrete = V horieť -- it burns, one lights it (Holz, Kerze, Papier, Kohle, Zigarette, Feuer)
foldable : concrete = V zložiť -- one folds it (Papier, Brief, Hemd, Handtuch, Karte, Decke, Zeitung)
cuttable : concrete = V krájať -- one cuts, slices, peels it (Brot, Kuchen, Apfel, Papier, Haare, Fleisch, Stoff)
switchable : concrete = V zapnúť (prístroj) -- one switches it on and off (Licht, Lampe, Radio, Fernseher, Computer, Herd, Motor)
enterable : concrete = V vstúpiť -- one goes into it (Haus, Zimmer, Laden, Auto, Zug, Wald, Höhle, Garten)
climbable : concrete = V liezť -- one climbs it (Berg, Baum, Leiter, Treppe, Mauer, Turm)
fillable : concrete = V naplniť -- one fills and empties it (Glas, Flasche, Tasse, Eimer, Tank, Kiste)
attendable : abstract = V zúčastniť sa -- an occasion one takes part in, goes to, cancels (Treffen, Konzert, Kurs, Sitzung, Party, Spiel, Prüfung)
